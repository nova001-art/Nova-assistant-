<script>
const $ = id => document.getElementById(id);
const read = (key, fallback) => {
  try {
    const value = localStorage.getItem(key);
    return value ? JSON.parse(value) : fallback;
  } catch { return fallback; }
};
let messages = read("nova_messages_v1", [
  { role: "nova", text: "Hi, I'm Nova ✨ This is our first little space. I'm a prototype for now, but you can chat with me and save memories in the notebook. What would you like to explore?" }
]);
let memories = read("nova_memories_v1", []);

function store() {
  try {
    localStorage.setItem("nova_messages_v1", JSON.stringify(messages));
    localStorage.setItem("nova_memories_v1", JSON.stringify(memories));
  } catch {
    alert("The browser couldn't save this data. Check available storage.");
  }
}
function drawMessages() {
  const area = $("messages");
  area.replaceChildren();
  messages.forEach(m => {
    const el = document.createElement("div");
    el.className = "message " + (m.role === "user" ? "user" : "nova");
    el.textContent = m.text;
    area.appendChild(el);
  });
  area.scrollTop = area.scrollHeight;
}
function drawMemories() {
  const list = $("memoryList");
  list.replaceChildren();
  if (!memories.length) {
    const p = document.createElement("p");
    p.className = "note";
    p.textContent = "No memories saved yet.";
    list.appendChild(p);
  }
  memories.forEach((memory, index) => {
    const card = document.createElement("div");
    card.className = "panel";
    card.style.marginTop = "12px";
    const p = document.createElement("p");
    p.textContent = memory;
    const del = document.createElement("button");
    del.className = "action danger";
    del.textContent = "Delete this memory";
    del.onclick = () => {
      memories.splice(index, 1);
      store(); drawMemories();
    };
    card.append(p, del);
    list.appendChild(card);
  });
}
function showTab(tab) {
  ["chat", "memory", "settings"].forEach(name => {
    $(name + "Page").classList.toggle("hidden", name !== tab);
    $(name + "Tab").classList.toggle("active", name === tab);
  });
  if (tab === "memory") drawMemories();
}
function saveMemory() {
  const value = $("memoryInput").value.trim();
  if (!value) return;
  memories.push(value);
  $("memoryInput").value = "";
  store(); drawMemories();
}
function makeReply(text) {
  const t = text.toLowerCase();
  if (t.includes("hello") || t.includes("hi")) {
    return "Hey there ✨ What's on your mind today?";
  }
  if (t.includes("your name")) {
    return "I'm Nova — your warm, curious, and thoughtful conversation companion.";
  }
  if (t.includes("remember")) {
    return memories.length
      ? "Your saved notebook includes: " + memories.join(" • ")
      : "I don't have any saved memories yet. You can add one in the Memory tab.";
  }
  if (t.includes("sad") || t.includes("lonely") || t.includes("stress")) {
    return "That sounds like a lot to carry. We can take it one step at a time. Would you like to tell me more, or would a small distraction help?";
  }
  if (t.includes("game") || t.includes("gaming")) {
    return "Gaming can be a fun way to unwind 🎮 What have you been playing lately?";
  }
  return "I'm here to explore that with you. My replies are still limited because this version doesn't have a real AI model connected yet. You can tell me more, and we can keep building Nova together.";
}
$("chatForm").addEventListener("submit", event => {
  event.preventDefault();
  const text = $("messageInput").value.trim();
  if (!text) return;
  messages.push({ role: "user", text });
  $("messageInput").value = "";
  messages.push({ role: "nova", text: makeReply(text) });
  store(); drawMessages();
});
function clearChat() {
  if (!confirm("Delete the saved conversation?")) return;
  messages = [];
  store(); drawMessages();
}
function clearMemories() {
  if (!confirm("Permanently delete all saved memories from this browser?")) return;
  memories = [];
  store(); drawMemories();
}
function exportData() {
  const data = JSON.stringify({ messages, memories }, null, 2);
  const blob = new Blob([data], { type: "application/json" });
  const url = URL.createObjectURL(blob);
  const a = document.createElement("a");
  a.href = url; a.download = "nova-backup.json"; a.click();
  URL.revokeObjectURL(url);
}
drawMessages();
